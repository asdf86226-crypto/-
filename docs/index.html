구글 시트에 연결 (실시간 공유)

이 설정은 대표자 한 사람만, 한 번이면 됩니다. 연결 후에는 팀원 모두가 대시보드에서 직접 입력하고, 그 내용이 구글 시트에 저장되어 함께 보입니다. 키·SQL 없이 아래 코드를 붙여넣고 배포 URL 하나만 가져오면 됩니다.

1구글 시트 만들고 스크립트 열기
  1. sheets.new 에서 빈 스프레드시트 생성
  2. 상단 메뉴 확장 프로그램 → Apps Script 클릭
  3. 편집기의 기존 코드를 모두 지우고, 아래 코드를 붙여넣은 뒤 저장(💾)

let s=ss.getSheetByName(SHEET); if(!s){ s=ss.insertSheet(SHEET); } if(s.getLastRow()===0){ s.appendRow(COLS); } return s; } function read_(){ const s=sh_(); const v=s.getDataRange().getValues(); const h=v[0]; return v.slice(1).map(r=>{ const o={}; h.forEach((k,i)=>{ let c=r[i]; if(c instanceof Date) c=Utilities.formatDate(c,Session.getScriptTimeZone(),'yyyy-MM-dd'); o[k]=c; }); return o; }) .filter(o=>o.id); } function json_(o){ return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function html_(m){ return HtmlService.createHtmlOutput('<meta charset="utf-8"><p style="font:16px sans-serif;padding:24px;line-height:1.6">'+m+'</p>'); } function doGet(e){ const p=e.parameter||{}; if(p.code) return kakaoCb_(p.code); if(p.kakao_setup) return kakaoStart_(p.key); const out={ok:true,rows:read_()}; if(p.callback) return ContentService.createTextOutput(p.callback+'('+JSON.stringify(out)+')').setMimeType(ContentService.MimeType.JAVASCRIPT); return json_(out); } function doPost(e){ const b=JSON.parse(e.postData.contents||'{}'); const s=sh_(); const v=s.getDataRange().getValues(), h=v[0], idc=h.indexOf('id'); if(b.action==='add'){ (b.items||[]).forEach(it=>{ const id=Utilities.getUuid(); const cd=Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy-MM-dd'); s.appendRow(COLS.map(k=> k==='id'?id : k==='createdAt'?cd : (it[k]!=null?it[k]:''))); }); try{ notify_(b.items||[]); }catch(err){} } else if(b.action==='remove'){ for(let i=1;i<v.length;i++){ if(v[i][idc]===b.id){ s.deleteRow(i+1); break; } } } else if(b.action==='clear'){ if(s.getLastRow()>1) s.deleteRows(2,s.getLastRow()-1); } return json_({ok:true}); } // ===== 카카오톡 '나에게 보내기' 접수 알림 ===== function kakaoStart_(key){ if(key) P.setProperty('KAKAO_KEY',key); const u='https://kauth.kakao.com/oauth/authorize?response_type=code&scope=talk_message' +'&client_id='+encodeURIComponent(P.getProperty('KAKAO_KEY')||'') +'&redirect_uri='+encodeURIComponent(ScriptApp.getService().getUrl()); return HtmlService.createHtmlOutput('<meta charset="utf-8"><div style="font:16px sans-serif;padding:24px;line-height:1.7">카카오톡 알림을 연결합니다.<br><a href="'+u+'" target="_top" style="display:inline-block;margin-top:14px;background:#FEE500;padding:12px 20px;border-radius:8px;text-decoration:none;font-weight:700;color:#000">카카오 로그인하고 동의하기</a></div>'); } function kakaoCb_(code){ const key=P.getProperty('KAKAO_KEY'); const r=UrlFetchApp.fetch('https://kauth.kakao.com/oauth/token',{method:'post',muteHttpExceptions:true, payload:{grant_type:'authorization_code',client_id:key,redirect_uri:ScriptApp.getService().getUrl(),code:code}}); const t=JSON.parse(r.getContentText()); if(t.refresh_token){ P.setProperty('KAKAO_RT',t.refresh_token); return html_('✅ 카카오톡 알림이 연결되었습니다. 이제 접수되면 카톡이 옵니다. 이 창은 닫으셔도 됩니다.'); } return html_('연결 실패: '+r.getContentText()); } function notify_(items){ const key=P.getProperty('KAKAO_KEY'), rt=P.getProperty('KAKAO_RT'); if(!key||!rt||!items.length) return; const r=UrlFetchApp.fetch('https://kauth.kakao.com/oauth/token',{method:'post',muteHttpExceptions:true, payload:{grant_type:'refresh_token',client_id:key,refresh_token:rt}}); const t=JSON.parse(r.getContentText()); if(t.refresh_token) P.setProperty('KAKAO_RT',t.refresh_token); if(!t.access_token) return; let text='🛒 구매요청 '+items.length+'건 접수\n'+items.map(it=> '· ['+(it.team||'')+'] '+(it.itemName||it.code||'')+' '+(it.qty||'')+'개 / 입고 '+(it.dueDate||'-')).join('\n'); }

1시작하기

✓ 화면 오른쪽 위를 먼저 확인하세요.
구글 시트 연결 초록색이면 정상입니다. 입력한 내용이 팀 전체에 공유됩니다.
! 로컬 저장 회색으로 보인다면 멈춰주세요.
공유가 안 되는 상태라 입력해도 담당자에게 전달되지 않습니다. 전달받은 링크로 다시 접속하거나 담당자에게 알려주세요.

4올린 뒤 확인하기

오른쪽 목록에 모든 팀의 요청이 함께 표시됩니다. 내가 올린 건 바로 나타나고, 다른 사람이 올린 내용을 보려면 오른쪽 위 ⟳ 새로고침을 눌러주세요. 현황판 오른쪽에 몇 시 기준인지 표시됩니다.

• 내 팀만 보기 — 목록 위 팀 이름(개발·자재·생산·품질)을 누르면 그 팀만 걸러 봅니다. 전체를 누르면 다시 모두 보입니다.
• 찾기·정렬 — 검색창에 품목코드나 품명을 넣으면 바로 찾고, 표 제목을 누르면 그 기준으로 정렬됩니다.
• 엑셀로 받기 — 목록 아래 ⬇ CSV 내보내기를 누르면 지금 보이는 목록을 파일로 받습니다.

5재고 · 판매량 함께 보기

품목코드를 입력하면 왼쪽 아래 품목 분석에 그 품목의 현재고 · 월평균 소요 · 소진 예상이 바로 나타납니다. 목록에서도 품명 아래에 같은 내용이 한 줄로 표시됩니다.

• 소진 예상 — 지금 재고로 몇 개월을 버틸 수 있는지입니다. 1개월 미만은 빨간색, 2개월 미만은 주황색으로 표시되니 수량을 정할 때 참고하세요. 요청 수량을 넣으면 그 수량을 받은 뒤 몇 개월이 되는지도 같이 보여줍니다.
• 월평균 소요가 어디서 나온 값인지도 아래에 적혀 있습니다. 상품·제품은 실제 판매량, 자재·반제품은 그 부품이 들어가는 제품의 판매량에서 BOM으로 역산한 값이며, 둘 다 해당하면 두 값을 더해 보여줍니다.

i 이 정보는 구매 담당자가 월 1회 ERP에서 갱신합니다. 화면 아래 ERP 기준 날짜를 확인하세요. BOM이 아직 등록되지 않은 품목은 소요량이 비어 있을 수 있습니다 — 그대로 요청하셔도 됩니다.

ERP 데이터 불러오기

ERP에서 내려받은 재고현황 · 판매현황 · BOM 파일을 한꺼번에 올리면 이 브라우저 안에서 바로 계산해 품목별 재고 · 월평균 소요량 · 소진 예상을 만들어 둡니다. 파일은 어디에도 전송되지 않습니다.

1ERP에서 파일 내보내기
  1. 재고현황 — 품목코드 · 품명 · 품목구분 · 재고수량 열이 포함되도록 내보내기
    플로우텍·태성정밀 각각 따로 올려도 됩니다
  2. 판매현황내역 — 최근 2년치(예: 25년 · 26년)
  3. BOM(소요량)현황 — 자재·반제품의 소요량을 역산하는 데 씁니다
2파일 올리기 (여러 개 한꺼번에)

    분석 결과는 내 브라우저에만 저장됩니다. 팀원 모두에게 같은 정보를 보이게 하려면 팀 공유용 파일 저장으로 받은 erp-data.json 을 담당자에게 전달해 대시보드와 같은 폴더에 올려두면 됩니다.
