구글 시트에 연결 (실시간 공유)

이 설정은 대표자 한 사람만, 한 번이면 됩니다. 연결 후에는 팀원 모두가 대시보드에서 직접 입력하고, 그 내용이 구글 시트에 저장되어 함께 보입니다. 키·SQL 없이 아래 코드를 붙여넣고 배포 URL 하나만 가져오면 됩니다.

1구글 시트 만들고 스크립트 열기
  1. sheets.new 에서 빈 스프레드시트 생성
  2. 상단 메뉴 확장 프로그램 → Apps Script 클릭
  3. 편집기의 기존 코드를 모두 지우고, 아래 코드를 붙여넣은 뒤 저장(💾)
const SHEET = 'requests'; const COLS = ['id','createdAt','team','code','itemName','qty','dueDate','requester','note']; const P = PropertiesService.getScriptProperties(); function sh_(){ const ss=SpreadsheetApp.getActiveSpreadsheet(); let s=ss.getSheetByName(SHEET); if(!s){ s=ss.insertSheet(SHEET); } if(s.getLastRow()===0){ s.appendRow(COLS); } return s; } function read_(){ const s=sh_(); const v=s.getDataRange().getValues(); const h=v[0]; return v.slice(1).map(r=>{ const o={}; h.forEach((k,i)=>{ let c=r[i]; if(c instanceof Date) c=Utilities.formatDate(c,Session.getScriptTimeZone(),'yyyy-MM-dd'); o[k]=c; }); return o; }) .filter(o=>o.id); } function json_(o){ return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function html_(m){ return HtmlService.createHtmlOutput('<meta charset="utf-8"><p style="font:16px sans-serif;padding:24px;line-height:1.6">'+m+'</p>'); } function doGet(e){ const p=e.parameter||{}; if(p.code) return kakaoCb_(p.code); if(p.kakao_setup) return kakaoStart_(p.key); const out={ok:true,rows:read_()}; if(p.callback) return ContentService.createTextOutput(p.callback+'('+JSON.stringify(out)+')').setMimeType(ContentService.MimeType.JAVASCRIPT); return json_(out); } function doPost(e){ const b=JSON.parse(e.postData.contents||'{}'); const s=sh_(); const v=s.getDataRange().getValues(), h=v[0], idc=h.indexOf('id'); if(b.action==='add'){ (b.items||[]).forEach(it=>{ const id=Utilities.getUuid(); const cd=Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy-MM-dd'); s.appendRow(COLS.map(k=> k==='id'?id : k==='createdAt'?cd : (it[k]!=null?it[k]:''))); }); try{ notify_(b.items||[]); }catch(err){} } else if(b.action==='remove'){ for(let i=1;i<v.length;i++){ if(v[i][idc]===b.id){ s.deleteRow(i+1); break; } } } else if(b.action==='clear'){ if(s.getLastRow()>1) s.deleteRows(2,s.getLastRow()-1); } return json_({ok:true}); } // ===== 카카오톡 '나에게 보내기' 접수 알림 ===== function kakaoStart_(key){ if(key) P.setProperty('KAKAO_KEY',key); const u='https://kauth.kakao.com/oauth/authorize?response_type=code&scope=talk_message' +'&client_id='+encodeURIComponent(P.getProperty('KAKAO_KEY')||'') +'&redirect_uri='+encodeURIComponent(ScriptApp.getService().getUrl()); return HtmlService.createHtmlOutput('<meta charset="utf-8"><div style="font:16px sans-serif;padding:24px;line-height:1.7">카카오톡 알림을 연결합니다.<br><a href="'+u+'" target="_top" style="display:inline-block;margin-top:14px;background:#FEE500;padding:12px 20px;border-radius:8px;text-decoration:none;font-weight:700;color:#000">카카오 로그인하고 동의하기</a></div>'); } function kakaoCb_(code){ const key=P.getProperty('KAKAO_KEY'); const r=UrlFetchApp.fetch('https://kauth.kakao.com/oauth/token',{method:'post',muteHttpExceptions:true, payload:{grant_type:'authorization_code',client_id:key,redirect_uri:ScriptApp.getService().getUrl(),code:code}}); const t=JSON.parse(r.getContentText()); if(t.refresh_token){ P.setProperty('KAKAO_RT',t.refresh_token); return html_('✅ 카카오톡 알림이 연결되었습니다. 이제 접수되면 카톡이 옵니다. 이 창은 닫으셔도 됩니다.'); } return html_('연결 실패: '+r.getContentText()); } function notify_(items){ const key=P.getProperty('KAKAO_KEY'), rt=P.getProperty('KAKAO_RT'); if(!key||!rt||!items.length) return; const r=UrlFetchApp.fetch('https://kauth.kakao.com/oauth/token',{method:'post',muteHttpExceptions:true, payload:{grant_type:'refresh_token',client_id:key,refresh_token:rt}}); const t=JSON.parse(r.getContentText()); if(t.refresh_token) P.setProperty('KAKAO_RT',t.refresh_token); if(!t.access_token) return; let text='🛒 구매요청 '+items.length+'건 접수\n'+items.map(it=> '· ['+(it.team||'')+'] '+(it.itemName||it.code||'')+' '+(it.qty||'')+'개 / 입고 '+(it.dueDate||'-')).join('\n'); if(text.length>190) text=text.substring(0,188)+'…'; const tmpl={object_type:'text',text:text,link:{web_url:'https://kakao.com'}}; UrlFetchApp.fetch('https://kapi.kakao.com/v2/api/talk/memo/default/send',{method:'post',muteHttpExceptions:true, headers:{Authorization:'Bearer '+t.access_token},payload:{template_object:JSON.stringify(tmpl)}}); }
2웹앱으로 배포
  1. 우측 상단 배포 → 새 배포 → 유형 웹 앱
  2. 실행: 나(본인) · 액세스 권한: 모든 사용자 로 설정 → 배포
  3. 권한 요청 시 본인 구글 계정 승인 (고급 → 이동 눌러 진행)
  4. 표시된 웹 앱 URL(…/exec 로 끝남) 복사
🔔카카오톡 접수 알림 (선택)

접수가 등록되면 대표자 본인 카카오톡으로 알림이 옵니다. 아래 설정도 대표자만 1회.

  1. developers.kakao.com → 애플리케이션 추가하기로 앱 생성
  2. 왼쪽 앱 키에서 REST API 키 복사
  3. 왼쪽 카카오 로그인 → 활성화 설정 ON, Redirect URI에 아래 주소 등록:
    먼저 위에서 시트를 연결하세요
  4. 카카오 로그인 → 동의항목에서 카카오톡 메시지 전송을 '이용 중 동의'로 설정
  5. 아래에 REST API 키를 넣고 카카오톡 연결 → 열린 창에서 로그인·동의하면 완료

※ 코드를 새로 붙여넣었으면 배포 → 배포 관리 → 편집 → 새 버전으로 다시 배포하고, 실행 권한 재승인(외부 서비스 연결) 후 이용하세요.

1시작하기

전달받은 링크를 열면 바로 사용할 수 있습니다. 즐겨찾기에 추가해 두면 편합니다.

✓ 화면 오른쪽 위를 먼저 확인하세요.
구글 시트 연결 초록색이면 정상입니다. 입력한 내용이 팀 전체에 공유됩니다.
! 로컬 저장 회색으로 보인다면 멈춰주세요.
공유가 안 되는 상태라 입력해도 담당자에게 전달되지 않습니다. 전달받은 링크로 다시 접속하거나 담당자에게 알려주세요.

3여러 건 올리기 — 엑셀 파일

이미 엑셀로 정리해 두셨다면 엑셀 파일 탭에서 파일을 올리시면 됩니다. 붙여넣기가 아니라 파일을 그대로 선택하거나 끌어다 놓으면 자동으로 읽어옵니다.

올리는 순서 ① 요청팀 선택 → ② 파일 선택(또는 끌어다 놓기) → ③ “○건 인식” 확인 → ④ 인식된 항목 일괄 추가

열 순서는 아래와 같아야 합니다. 맨 윗줄 제목은 자동으로 걸러내니 그대로 두셔도 됩니다.

4올린 뒤 확인하기

오른쪽 목록에 모든 팀의 요청이 함께 표시됩니다. 내가 올린 건 바로 나타나고, 다른 사람이 올린 내용을 보려면 오른쪽 위 ⟳ 새로고침을 눌러주세요. 현황판 오른쪽에 몇 시 기준인지 표시됩니다.

• 내 팀만 보기 — 목록 위 팀 이름(개발·자재·생산·품질)을 누르면 그 팀만 걸러 봅니다. 전체를 누르면 다시 모두 보입니다.
• 찾기·정렬 — 검색창에 품목코드나 품명을 넣으면 바로 찾고, 표 제목을 누르면 그 기준으로 정렬됩니다.
• 엑셀로 받기 — 목록 아래 ⬇ CSV 내보내기를 누르면 지금 보이는 목록을 파일로 받습니다.

5주의사항

⚠ 삭제는 모두에게 적용됩니다. 각 줄 끝의 ✕를 누르면 그 요청이 모든 사람의 화면에서 사라집니다. 내가 올린 것이 아니라면 지우지 마시고, 잘못 올렸을 때만 본인 것을 지워주세요. 아래쪽 전체 삭제는 되돌릴 수 없으니 누르지 마세요.
! 수정 기능은 아직 없습니다. 잘못 올렸다면 그 줄을 지우고 다시 올려주세요.
i 요청이 접수되면 구매 담당자에게 알림이 갑니다. 올리신 뒤 따로 연락하지 않으셔도 됩니다. 급한 건은 기타사항에 “긴급”이라고 적어주세요.

품목코드를 모르면 어떻게 하나요?
