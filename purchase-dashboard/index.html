구글 시트에 연결 (실시간 공유)

이 설정은 대표자 한 사람만, 한 번이면 됩니다. 연결 후에는 팀원 모두가 대시보드에서 직접 입력하고, 그 내용이 구글 시트에 저장되어 함께 보입니다. 키·SQL 없이 아래 코드를 붙여넣고 배포 URL 하나만 가져오면 됩니다.

1구글 시트 만들고 스크립트 열기
  1. sheets.new 에서 빈 스프레드시트 생성
  2. 상단 메뉴 확장 프로그램 → Apps Script 클릭
  3. 편집기의 기존 코드를 모두 지우고, 아래 코드를 붙여넣은 뒤 저장(💾)
const SHEET = 'requests'; const COLS = ['id','createdAt','team','code','itemName','qty','dueDate','requester','note']; function sh_(){ const ss=SpreadsheetApp.getActiveSpreadsheet(); let s=ss.getSheetByName(SHEET); if(!s){ s=ss.insertSheet(SHEET); } if(s.getLastRow()===0){ s.appendRow(COLS); } return s; } function read_(){ const s=sh_(); const v=s.getDataRange().getValues(); const h=v[0]; return v.slice(1).map(r=>{ const o={}; h.forEach((k,i)=>{ let c=r[i]; if(c instanceof Date) c=Utilities.formatDate(c,Session.getScriptTimeZone(),'yyyy-MM-dd'); o[k]=c; }); return o; }) .filter(o=>o.id); } function doGet(e){ const out={ok:true,rows:read_()}; const cb=e&&e.parameter&&e.parameter.callback; if(cb) return ContentService.createTextOutput(cb+'('+JSON.stringify(out)+')').setMimeType(ContentService.MimeType.JAVASCRIPT); return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } function doPost(e){ const b=JSON.parse(e.postData.contents||'{}'); const s=sh_(); const v=s.getDataRange().getValues(), h=v[0], idc=h.indexOf('id'); if(b.action==='add'){ (b.items||[]).forEach(it=>{ const id=Utilities.getUuid(); const cd=Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy-MM-dd'); s.appendRow(COLS.map(k=> k==='id'?id : k==='createdAt'?cd : (it[k]!=null?it[k]:''))); }); } else if(b.action==='remove'){ for(let i=1;i1) s.deleteRows(2,s.getLastRow()-1); } return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); }
2웹앱으로 배포
  1. 우측 상단 배포 → 새 배포 → 유형 웹 앱
  2. 실행: 나(본인) · 액세스 권한: 모든 사용자 로 설정 → 배포
  3. 권한 요청 시 본인 구글 계정 승인 (고급 → 이동 눌러 진행)
  4. 표시된 웹 앱 URL(…/exec 로 끝남) 복사

엑셀(.xlsx) 또는 CSV 파일을 올리면 자동으로 읽어 등록합니다. 열 순서: 품목코드 · 품명 · 수량 · 입고요청일 · 기타사항
